01What is ES6?
Definition
ES6 (ECMAScript 2015) is the 6th version of JavaScript released in 2015. It makes code shorter, cleaner, and easier to maintain.
Before vs After ES6
Before ES6
var name = "Pravin"; console.log("Hello " + name);
After ES6
let name = "Pravin"; console.log(`Hello ${name}`);
Key Features
let & constArrow FunctionsDefault ParametersTemplate LiteralsDestructuringSpread OperatorRest OperatorClassesModulesPromises
Interview Answer
ES6 stands for ECMAScript 2015 — the sixth version of JavaScript. It introduced let, const, arrow functions, template literals, destructuring, classes, modules, promises, and more to make code cleaner and easier to maintain.
02Default Parameters
Definition
Default Parameters let us assign a fallback value to a function parameter when no value is passed. Without this, JavaScript returns undefined.
Problem without default parameters
function greet(name) { console.log("Hello " + name); } greet();
Output
Hello undefined
Solution with default parameters
function greet(name = "Guest") { console.log("Hello " + name); } greet(); greet("Pravin");
Output
Hello Guest Hello Pravin
Example — Employee Salary
function employee(name = "Employee", salary = 15000) { console.log(name + " Salary: " + salary); } employee(); employee("Pravin"); employee("Pravin", 50000);
Output
Employee Salary: 15000 Pravin Salary: 15000 Pravin Salary: 50000
Real-life Example — Electricity Bill
function bill(amount, tax = 18) { let total = amount + (amount * tax / 100); console.log("Total Bill = " + total); } bill(1000); bill(1000, 10);
Output
Total Bill = 1180 Total Bill = 1100
Important — undefined vs null
Passing undefined → default value is used.
Passing null → null is used (treated as an actual value).
function demo(x = 10) { console.log(x); } demo(undefined); // 10 demo(null); // null
03Spread Operator ( ... )
Definition
The Spread Operator ( ... ) expands elements of an array or object into individual values.
Basic — how it works
Without spread
let n = [10,20,30]; console.log(n);
Output
[10, 20, 30]
With spread
let n = [10,20,30]; console.log(...n);
Output
10 20 30
Copying an array
let arr1 = [10, 20, 30]; let arr2 = [...arr1]; // new copy arr2.push(40); console.log(arr1); // [10,20,30] ← unchanged console.log(arr2); // [10,20,30,40]
Merging arrays
let arr1 = [1,2,3]; let arr2 = [4,5,6]; let merged = [...arr1, ...arr2]; console.log(merged); // [1,2,3,4,5,6]
Passing array to a function
function add(a,b,c) { return a+b+c; } let numbers = [10,20,30]; console.log(add(...numbers)); // 60
Spread with objects
let basic = { name:"Pravin", age:22 }; let job = { role:"Developer", salary:50000 }; let emp = { ...basic, ...job }; console.log(emp);
Output
{ name:"Pravin", age:22, role:"Developer", salary:50000 }
Memory Trick — SPREAD = EXPAND  ·  Array → Individual Values
04Rest Parameter ( ... )
Definition
The Rest Parameter ( ... ) collects multiple arguments passed to a function into a single array. Used inside function parameters.
Basic example
function demo(...numbers) { console.log(numbers); } demo(10, 20, 30, 40);
Output
[10, 20, 30, 40]
Sum of any number of values
function sum(...numbers) { let total = 0; for(let num of numbers) { total += num; } return total; } console.log(sum(10, 20, 30, 40)); // 100
Mixed — normal + rest parameter
function student(course, ...names) { console.log(course); console.log(names); } student("Java", "Pravin", "Ramesh", "Siva");
Output
Java ["Pravin", "Ramesh", "Siva"]
Important Rule: Rest parameter must always be the LAST parameter. Writing it before other params causes an error.
Rest vs Spread comparison
Rest ParameterSpread Operator
Collects valuesExpands values
Used in function paramsUsed when calling functions / with arrays
Many values → one arrayArray → individual values
Packs dataUnpacks data
Memory Trick — REST = COLLECT  ·  Individual Values → Array
05Destructuring
Definition
Destructuring is a shortcut to extract values from arrays or objects into individual variables in one clean line.
Array destructuring — basic
let fruits = ["Apple", "Banana", "Mango"]; let [a, b, c] = fruits; console.log(a); // Apple console.log(b); // Banana console.log(c); // Mango
Skipping values
let colors = ["Red", "Green", "Blue"]; let [first, , third] = colors; console.log(first); // Red console.log(third); // Blue
Swapping variables
let a = 10, b = 20; [a, b] = [b, a]; console.log(a); // 20 console.log(b); // 10
Default values in array destructuring
let numbers = [10]; let [a, b = 20] = numbers; console.log(a); // 10 console.log(b); // 20 ← default used
Object destructuring — basic
let student = { name:"Pravin", age:22, course:"Java" }; let { name, age, course } = student; console.log(name); // Pravin console.log(age); // 22 console.log(course); // Java
For object destructuring — variable names MUST match property names, otherwise the value will be undefined.
Renaming variables
let student = { name:"Pravin", age:22 }; let { name: studentName, age: studentAge } = student; console.log(studentName); // Pravin console.log(studentAge); // 22
Default values in object destructuring
let student = { name:"Pravin" }; let { name, course = "Java" } = student; console.log(name); // Pravin console.log(course); // Java ← default used
Destructuring in function parameters
Without
function show(s){ console.log(s.name); console.log(s.age); }
With
function show({name,age}){ console.log(name); console.log(age); }
06ES6 Modules
Definition
A Module is a separate JavaScript file that holds specific functionality and can be imported into any other file whenever needed. Keeps projects clean and organised.
Module flow
Create function→Export it→Import it→Use it
Named export
// math.js export const pi = 3.14; export function add(a,b){ return a+b; } export function sub(a,b){ return a-b; }
// app.js import { pi, add, sub } from "./math.js"; console.log(pi); // 3.14 console.log(add(10,20)); // 30 console.log(sub(20,10)); // 10
Default export
// greet.js export default function greet(){ console.log("Welcome"); }
// app.js import greet from "./greet.js"; // no { } needed greet(); // Welcome
Using both together
// student.js export const course = "Java"; export function add(a,b){ return a+b; } export default function greet(){ console.log("Hello Student"); }
// app.js import greet, { course, add } from "./student.js"; greet(); console.log(course); // Java console.log(add(10,20)); // 30
Named vs Default — comparison
FeatureNamed ExportDefault Export
How many per file?Multiple allowedOnly one
Curly braces?Yes — requiredNo — not needed
Syntax (export)export { add }export default add
Syntax (import)import { add }import add
To use ES6 Modules in HTML add type="module" to the script tag — otherwise imports and exports will not work.
